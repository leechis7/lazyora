func (g gridProbeTerminal) Colors() int { return g.count }
func gridProbeFixture(t *testing.T, theme string, colors int) (*screen, tuivory.SimulationScreen) {
	raw := tuivory.NewSimulationScreen("UTF-8")
	if err := raw.Init(); err != nil {
		t.Fatal(err)
	}
	raw.SetSize(120, 36)
	t.Cleanup(raw.Fini)
	terminal := gridProbeTerminal{raw, colors}
	app := tuivory.NewApplication().SetScreen(terminal)
	s := newWorkspaceScreen(app, terminal, settings.Config{Language: "ko", Theme: theme, DisplayMode: "unicode", Mouse: true}, filepath.Join(t.TempDir(), "settings.json"), true, nil)
	spaces := &workspaceSet{app: app, active: s}
	spaces.adopt(s)
	t.Cleanup(func() { s.workerClosed = false; spaces.shutdown() })
	s.workerClosed = true // Never submit SQL or connect during this display/input audit.
	s.connected = true
	data := oracle.Result{EditID: 5, EditObject: oracle.Object{Owner: "AUDIT", Name: "ITEMS", Kind: "TABLE"}, EditRowIDColumn: 1, EditColumns: []bool{false, true, true, true, true}, EditColumnNames: []string{"", "ID", "TITLE", "AMOUNT", "NOTE"}, Columns: []string{"RID", "ID", "TITLE", "AMOUNT", "NOTE"}}
	for i := 0; i < 30; i++ {
		id := fmt.Sprint(i + 1)
		title := "한글 " + id
		note := "note " + id
		if i == 1 {
			title = "long\t[red]\n" + strings.Repeat("가", 90)
		}
		null := i == 0
		cellNote := note
		if null {
			note = "NULL"
			cellNote = ""
		}
		data.EditRowIDs = append(data.EditRowIDs, "r"+id)
		data.Rows = append(data.Rows, []string{"r" + id, id, title, "12345.67", note})
		data.Values = append(data.Values, []oracle.Cell{{Text: "r" + id, Type: "ROWID"}, {Text: id, Type: "NUMBER"}, {Text: title, Type: "VARCHAR2"}, {Text: "12345.67", Type: "NUMBER"}, {Text: cellNote, Type: "VARCHAR2", Null: null}})
	}
	s.render(data)
	for c, w := range []int{8, 18, 14, 18} {
		s.table.Select(1, c)
		s.resizeResultColumn(w - s.grid.widths[c])
	}
	s.table.Select(1, 1)
	app.SetFocus(s.table)
	app.ForceDraw()
	return s, raw
}
func gridProbeKey(s *screen, e *tuivory.EventKey) {
	if f := s.app.GetInputCapture(); f != nil {
		e = f(e)
	}
	if e != nil {
		if p := s.app.GetFocus(); p != nil {
			if f := p.InputHandler(); f != nil {
				f(e, func(p tuivory.Primitive) { s.app.SetFocus(p) })
			}
		}
	}
	s.app.ForceDraw()
}
func gridProbeMouse(s *screen, a tuivory.MouseAction, x, y int, b tuivory.ButtonMask) {
	e := tuivory.NewEventMouse(x, y, b, 0)
	if f := s.app.GetMouseCapture(); f != nil {
		e, a = f(e, a)
	}
	if e != nil {
		s.rootLayout.MouseHandler()(a, e, func(p tuivory.Primitive) { s.app.SetFocus(p) })
	}
	s.app.ForceDraw()
}
func gridProbeStyle(s *screen, raw tuivory.SimulationScreen, row, col int) map[string]any {
	x, y, w := s.resultCellRect(row, col)
	if x < 0 || y < 0 || w < 1 {
		return map[string]any{"visible": false}
	}
	r, _, st, _ := raw.GetContent(x+min(1, w-1), y)
	fg, bg, a := st.Decompose()
	return map[string]any{"x": x, "y": y, "width": w, "glyph": string(r), "fg": fmt.Sprintf("#%06x", fg.Hex()), "bg": fmt.Sprintf("#%06x", bg.Hex()), "underline": a&tuivory.AttrUnderline != 0, "italic": a&tuivory.AttrItalic != 0, "bold": a&tuivory.AttrBold != 0}
}
func gridProbeDump(t *testing.T, raw tuivory.SimulationScreen, name string) {
	t.Helper()
	dir := os.Getenv("GRID_AUDIT_DIR")
	if dir == "" {
		return
	}
	cells, w, h := raw.GetContents()
	out := []map[string]any{}
	var text strings.Builder
	for y := 0; y < h; y++ {
		for x := 0; x < w; {
			c := cells[y*w+x]
			v := string(c.Runes)
			if v == "" {
				v = " "
			}
			fg, bg, a := c.Style.Decompose()
			width := max(1, tuivory.TaggedStringWidth(v))
			out = append(out, map[string]any{"X": x, "Y": y, "W": width, "Text": v, "FG": fmt.Sprintf("#%06x", fg.Hex()), "BG": fmt.Sprintf("#%06x", bg.Hex()), "Bold": a&tuivory.AttrBold != 0, "Underline": a&tuivory.AttrUnderline != 0, "Italic": a&tuivory.AttrItalic != 0})
			text.WriteString(v)
			x += width
		}
		text.WriteByte('\n')
	}
	_ = os.MkdirAll(dir, 0755)
	b, _ := json.Marshal(map[string]any{"width": w, "height": h, "cells": out})
	if err := os.WriteFile(filepath.Join(dir, name+".json"), b, 0600); err != nil {
		t.Fatal(err)
	}
	_ = os.WriteFile(filepath.Join(dir, name+".txt"), []byte(text.String()), 0600)
}
